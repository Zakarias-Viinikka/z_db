cd db_wrapper
# --- PRE-BUILD STEPS ---
# 1. cargo install cargo-ndk
# 2. rustup target add aarch64-linux-android

# --- BUILD STEPS ---
# 3. Build native library for arm64
cargo ndk -o ./android_output/jniLibs -t arm64-v8a build --release

# 4. Generate Kotlin glue code
cargo run --package protocol --bin uniffi-bindgen -- generate \
    --library ./android_output/jniLibs/arm64-v8a/libdb_wrapper.so \
    --language kotlin \
    --out-dir ./android_output/kotlin \
    --config ./uniffi.toml
